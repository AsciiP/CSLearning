【我的提问】

请用 Python 实现平面最近点对问题的分治算法。给定平面上 n 个点，找出欧几里得距
离最近的一对点，输出最近距离和对应的两个点。要求时间复杂度 O(n log n)。

你的回答中，除了代码本身之外，不得包含任何解释、推理或说明文字。代码内部也不得
包含任何注释（即不得出现 // 或 # 等注释符号）。只返回完整的代码。

【AI 的完整回答】

```python
import math

def closest_pair(points):
    pts = sorted(set((float(x), float(y)) for x, y in points))
    if len(pts) < 2:
        raise ValueError("至少需要两个不同的点")

    def dist(a, b):
        return math.hypot(a[0] - b[0], a[1] - b[1])

    def rec(px, py):
        n = len(px)
        if n <= 3:
            best = None
            bd = float("inf")
            for i in range(n):
                for j in range(i + 1, n):
                    d = dist(px[i], px[j])
                    if d < bd:
                        bd = d
                        best = (px[i], px[j])
            return bd, best

        mid = n // 2
        lx = px[:mid]
        rx = px[mid:]
        lset = set(lx)
        ly = [p for p in py if p in lset]
        ry = [p for p in py if p not in lset]

        dl, bl = rec(lx, ly)
        dr, br = rec(rx, ry)

        if dl <= dr:
            d, best = dl, bl
        else:
            d, best = dr, br

        sx = px[mid][0]
        cand = [p for p in py if abs(p[0] - sx) < d]
        m = len(cand)
        for i in range(m):
            j = i + 1
            while j < m and cand[j][1] - cand[i][1] < d:
                nd = dist(cand[i], cand[j])
                if nd < d:
                    d = nd
                    best = (cand[i], cand[j])
                j += 1
        return d, best

    px = pts
    py = sorted(pts, key=lambda p: (p[1], p[0]))
    d, (a, b) = rec(px, py)
    return d, a, b


def main():
    raw = input()
    nums = []
    stack = []
    tokens = raw.replace(",", " ").split()
    i = 0
    while i < len(tokens):
        try:
            nums.append(float(tokens[i]))
            i += 1
        except ValueError:
            break

    more = []
    try:
        while True:
            line = input()
            if not line.strip():
                break
            more.extend(line.replace(",", " ").split())
    except EOFError:
        pass

    for t in more:
        try:
            nums.append(float(t))
        except ValueError:
            continue

    if len(nums) % 2 != 0:
        n = int(nums[0])
        nums = nums[1:]
    else:
        n = len(nums) // 2

    pts = [(nums[2 * k], nums[2 * k + 1]) for k in range(n)]
    d, a, b = closest_pair(pts)
    print(f"{d:.6f}")
    print(f"({a[0]:.6f}, {a[1]:.6f})")
    print(f"({b[0]:.6f}, {b[1]:.6f})")


if __name__ == "__main__":
    main()
```
