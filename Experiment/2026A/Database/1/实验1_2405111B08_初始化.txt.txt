-- 2. 清理旧数据
DROP TABLE IF EXISTS dbo.ReadingReservation;
DROP TABLE IF EXISTS dbo.ReadingMeeting;
DROP TABLE IF EXISTS dbo.LibraryReader;

-- 3. 建表与约束
CREATE TABLE dbo.LibraryReader (
    ReaderID CHAR(4) PRIMARY KEY,
    ReaderName NVARCHAR(30),
    ReaderGroup NVARCHAR(50),
    ReadingInterest NVARCHAR(20)
);

CREATE TABLE dbo.ReadingMeeting (
    MeetingID CHAR(4) PRIMARY KEY,
    MeetingTheme NVARCHAR(60),
    MeetingKind NVARCHAR(12),
    MeetingRoom NVARCHAR(20),
    SeatCount INT,
    VacantSeats INT,
    CONSTRAINT CK_ReadingMeeting_Seats CHECK (SeatCount > 0 AND VacantSeats >= 0 AND VacantSeats <= SeatCount)
);

CREATE TABLE dbo.ReadingReservation (
    ReservationID CHAR(5) PRIMARY KEY,
    ReaderID CHAR(4),
    MeetingID CHAR(4),
    GuestNumber INT,
    ReserveOn DATE,
    ReservationStatus NVARCHAR(10),
    CONSTRAINT FK_Reservation_Reader FOREIGN KEY (ReaderID) REFERENCES dbo.LibraryReader(ReaderID),
    CONSTRAINT FK_Reservation_Meeting FOREIGN KEY (MeetingID) REFERENCES dbo.ReadingMeeting(MeetingID),
    CONSTRAINT CK_ReadingReservation_Data CHECK (GuestNumber > 0 AND ReservationStatus IN ('已报名', '已取消'))
);
